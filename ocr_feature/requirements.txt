# mAIstra OCR backend — direct dependencies only.
# We intentionally do NOT pin every transitive dependency (a full `pip freeze`
# pinned macOS-only versions like multidict==6.7.1 / aiohttp==3.14.2 that have
# no Windows wheels). Listing only direct deps lets pip resolve the correct
# platform-specific wheels on each machine (Windows / macOS / Linux).
#
# The old frozen list is kept as requirements.full.txt for reference.

# Core OCR engine + models (PaddleOCR runs on the PaddlePaddle framework,
# which is NOT auto-installed by paddleocr and must be listed explicitly).
#
# Pinned to 3.2.2, NOT 3.3.x: PaddlePaddle 3.3.0/3.3.1 has a confirmed
# regression in its PIR-to-oneDNN conversion path that crashes CPU inference
# on PP-OCRv5_server_det with
#   NotImplementedError: (Unimplemented) ConvertPirAttribute2RuntimeAttribute
#   not support [pir::ArrayAttribute<pir::DoubleAttribute>]
# Reproduced on Windows; see PaddlePaddle/Paddle#77340 and
# PaddlePaddle/PaddleOCR#18162. Verified working on 3.2.2 on both macOS
# (Apple Silicon) and Windows.
paddlepaddle==3.2.2
paddleocr==3.7.0

# Image preprocessing (headless: no GUI libs, avoids extra system deps)
opencv-python-headless
numpy

# FastAPI backend
fastapi
uvicorn
python-multipart
python-dotenv
requests

# Supabase client (used by the save-verified-text endpoint)
supabase
